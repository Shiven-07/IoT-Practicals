from flask import Flask

import RPi.GPIO as GPIO

app = Flask(__name__)

# GPIO Setup
GPIO.setmode(GPIO.BCM)

LIGHT = 17

FAN = 27

GPIO.setup(LIGHT, GPIO.OUT)

GPIO.setup(FAN, GPIO.OUT)

GPIO.output(LIGHT, GPIO.LOW)

GPIO.output(FAN, GPIO.LOW)


@app.route('/')
def home():

    <a href="/fan/on">ON</a><br><br>

    <a href="/fan/off">OFF</a>
    """


@app.route('/light/on')
def light_on():
    GPIO.output(LIGHT, GPIO.HIGH)
    return "Light ON"


@app.route('/light/off')
def light_off():
    GPIO.output(LIGHT, GPIO.LOW)
    return "Light OFF"


@app.route('/fan/on')
def fan_on():
