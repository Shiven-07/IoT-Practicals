# LCD configuration
lcd = CharLCD('PCF8574', 0x27)
lcd.clear()
lcd.write_string('Place Finger')

try:

    # Fingerprint sensor
    f = PyFingerprint('/dev/serial0', 57600, 0xFFFFFFFF,
                      0x00000000)

    if not f.verifyPassword():
        raise ValueError ('Fingerprint sensor password incorrect')

except Exception as e:

    print ('Sensor initialization failed!')

    print (e)

    exit (1)

while True:

    lcd.clear ()

    lcd.write_string ('Place Finger')

    # Wait for finger
    while not f.readImage():

        pass

    # Convert image

--- Original PDF page 50 ---
f.convertImage(0x01)

    # Search template

    result = f.searchTemplate()

    positionNumber = result[0]

    lcd.clear ()

    if positionNumber >= 0:

        lcd.write_string('Fingerprint')

        lcd.crlf ()

        lcd.write_string('Matched')

        print ("Fingerprint Matched")

    else:

        lcd.write_string('Fingerprint')

        lcd.crlf()

        lcd.write_string('Unmatched')

        print("Fingerprint Unmatched")

    sleep(3)
