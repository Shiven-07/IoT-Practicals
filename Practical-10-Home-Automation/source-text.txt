app = Flask(name)

@app.route('/light/off')
def light_off():
GPIO.output(LIGHT, GPIO.LOW)
return "Light OFF"

@app.route('/fan/on')
def fan_on():
GPIO.output(FAN, GPIO.HIGH)
return "Fan ON"

@app.route('/fan/off')
def fan_off():
GPIO.output (FAN, GPIO.LOW)

if name ==' _main':
app.run(host='0.0.0.0', port=5000)
